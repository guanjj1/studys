### docker-compose.yaml编写
#### 常用命令
```shell
#在 docker-compose build 命令中添加 --no-cache 选项，以确保每次都从头开始构建镜像：
docker compose build --no-cache
```

#### docker-compose.yaml
```yaml
version: "3" # 已过时
name: "fgbox" # 不写默认是文件夹名称
services: #服务名似乎不能大写
  appbackend:
    # 使用存在的镜像
    # image可使用本地镜像，而Dockerfile似乎不能
    image: golang:1.22.5-alpine3.20
    # 使用自定义镜像，与image冲突，
    build: 
      context: ./appBackend #
      dockerfile: appbackend #不写默认读取Dockerfile
    ports:
      - "9090:9090"
    volumes:
      - ./appBackend/files:/var/lib/mysql
      - ./appBackend/log:/etc/mysql
      - ./appBackend/config.yaml:/etc/mysql
      - /etc/localtime:/etc/localtime
    networks:
      - netfgbox
    # environment:
    #   - MYSQL_ROOT_PASSWORD=ewake666
    # # 建议使用env_file替代environment
    # env_file:
    #   - ./appDB/.mysql_env
    # 别的容器访问该容器使用的端口,可写在Dockerfile中
    expose:
      - "9090"
    depends_on: 
      - appredis # appredis启动完后启动
      - appdb
    restart: always
  appdb:
    # 使用存在的镜像
    # image可使用本地镜像，而Dockerfile似乎不能
    image: mysql:5.7
    # 使用自定义镜像，与image冲突，
    build: 
      context: ./appDB #
      dockerfile: appdb #不写默认读取Dockerfile
    ports:
      - "3306:3306"
    volumes:
      - ./appDB/data:/var/lib/mysql
      - ./appDB/conf:/etc/mysql
      - ./appDB/healthcheck.sh:/check/healthcheck.sh
      - /etc/localtime:/etc/localtime
    networks:
      - netfgbox
    environment:
      - MYSQL_ROOT_PASSWORD=ewake666
    # 建议使用env_file替代environment
    env_file:
      - ./appDB/.mysql_env
    # 别的容器访问该容器使用的端口,可写在Dockerfile中
    expose:
      - "3306"
    depends_on: 
      - appredis # appredis启动完后启动
    restart: always
    command: ["--max_connections=6000","--wait_timeout=60"]
    #确保启动成功
    healthcheck:
      # test: ["CMD", "mysqladmin" ,"ping", "-u", "root", "-p${MYSQL_ROOT_PASSWORD}"]
      #脚本例子在下面
      test: ./check/healthcheck.sh
      interval: 10s
      timeout: 5s
      retries: 5
  appredis: 
    build: 
      context: ./appRedis
    # 报错The requested image's platform (linux/arm64) does not match the detected host platform，尝试添加下面的（添加下面的并不能解决实际问题，还是得在对应平台拉取镜像）
    # platform: linux/arm64
    ports:
      - "6379:6379"
    volumes:
      - ./appRedis/data:/data
      - ./appRedis/conf/redis.conf:/etc/redis/redis.conf
      - /etc/localtime:/etc/localtime
    networks:
      - netfgbox
    # 可写在Dockerfile中
    expose:
      - "6379"
    restart: always
    # 可写在Dockerfile中,指定使用 /etc/redis/redis.conf 配置文件启动 Redis 服务
    command: ["redis-server", "/etc/redis/redis.conf"]
    #确保启动成功
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
  nginx:
    # image: nginx:alpine # 使用官方的 Nginx 镜像
    build: 
      context: ./nginx
    ports:
      - "80:80" # 将容器的 80 端口映射到主机的 80 端口（通常用于 web 服务）
    networks:
      - netfgbox
    volumes:
      - ./nginx/conf.d/default.conf:/etc/nginx/conf.d/default.conf
    depends_on: # 确保 Nginx 在启动时，Go 应用已经运行（可选）
      - appbackend
    expose:
      - "80"
    restart: always
    
networks:
# 网络要提前创建好 
  netfgbox:
```

#### docker创建网络
<https://blog.csdn.net/meidongyan/article/details/135905080>
```shell
# 连接容器到网络
docker network connect 网络名 

# 创建网络(使用统一网络的容器，肯定可以ping通)
docker network create  网络名 

# 断开网络
docker network disconnect  网络名

# 查看网络具体信息
docker network inspect  

# 查看所有网络列表
docker network ls   

# 删除无用网络
docker network prune  

# 删除指定的网络
docker network rm 

```

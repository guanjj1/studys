# 常用命令
![](images/img-2023-04-03-17-55-25.png)

```dockerfile
# Dockerfile
FROM golang:1.22.5 as builder

# 设置工作目录 无则自动创建
WORKDIR /app

# # 复制 go.mod 和 go.sum（如果使用模块）
# COPY go.mod go.sum ./

# # 下载依赖项
# RUN go mod download

# 复制项目文件
COPY ./src .

# 下载依赖项
RUN go mod tidy

# 构建应用
RUN go build -o fgbox


# 使用轻量级的 Alpine Linux 作为最终的运行环境
FROM alpine:latest
WORKDIR /opt/

# 从构建阶段复制二进制文件,目标路径不存在自动创建
COPY --from=builder /app/fgbox ./fgbox/

# 暴露端口
EXPOSE 8080

# 运行应用
CMD ["./fgbox/fgbox"]
```
